s = Slider(0.25, 2, 0.05, 1, 250, false, true, false, false)

lesson = Slider(1, 3, 1, 1, 250, false, true, false, false)
sphereStep = Slider(1, 4, 1, 1, 250, false, true, false, false)

SetValue(showSphere, true)

B = 11*s
h = 11*s/q
A = B/2
P1 = (-A, -A, 0)
P2 = (A, -A, 0)
P3 = (A, A, 0)
P4 = (-A, A, 0)
O = (0, 0, 0)
V = (0, 0, h)
M = Midpoint(P1, P2)
face1 = Polygon(P1, P2, V)
face2 = Polygon(P2, P3, V)
face3 = Polygon(P3, P4, V)
face4 = Polygon(P4, P1, V)
b1 = Segment(P1, P2)
b2 = Segment(P2, P3)
b3 = Segment(P3, P4)
b4 = Segment(P4, P1)
edge1 = Segment(P1, V)
edge2 = Segment(P2, V)
edge3 = Segment(P3, V)
edge4 = Segment(P4, V)
hSeg = Segment(O, V)
aSeg = Segment(O, M)
sSeg = Segment(M, V)
S = Length(sSeg)
phi = (1+sqrt(5))/2
r = A*h/(A+S)
I = (0, 0, r)
T = Dilate(V, A/S, M)
ball = Sphere(I, r)
secCircle = Circle(I, r, Vector((1, 0, 0)))
oiSeg = Segment(O, I)
ivSeg = Segment(I, V)
mtSeg = Segment(M, T)
tvSeg = Segment(T, V)
moSeg = Segment(M, O)
itSeg = Segment(I, T)
pf1 = Plane(P1, P2, V)
pf2 = Plane(P2, P3, V)
pf3 = Plane(P3, P4, V)
pf4 = Plane(P4, P1, V)
pb = Plane(P1, P2, P3)
dF1 = Distance(I, pf1)
dF2 = Distance(I, pf2)
dF3 = Distance(I, pf3)
dF4 = Distance(I, pf4)
dBase = Distance(I, pb)
IT = Distance(I, T)
OI = Length(oiSeg)
VI = Length(ivSeg)
MT = Length(mtSeg)
VT = Length(tvSeg)
MO = Length(moSeg)
VIoverOI = VI/OI
MToverVT = MT/VT
errVIOI = abs(VIoverOI-phi)/phi
errMTVT = abs(MToverVT-phi)/phi
t = Length(hSeg)/Length(aSeg)
k = sqrt(1+t^2)
limitSA = S/A
eggOK = t > 0 && z0 > 0 && z0^2 > 4*t
a = If(eggOK, (-z0+sqrt(z0^2-4*t))/(2*t))
b = If(eggOK, (-z0+sqrt(z0^2+4*t))/(2*t))
fy(x) = 1/(z0+t*x)^2 - x^2
gW(x) = x*(z0+t*x)^3 + t
xm = x(Root(gW, a, 0))
L = (b-a)*k
W = 2*sqrt(fy(xm))
LW = L/W
errLW = abs(LW-phi)/phi
errLimit = abs(limitSA-phi)/phi
auxLen = L/limitSA
g = 8/L
c = (a+b)/2
hw = (b-a)/2
oval = Curve(g*k*hw*cos(p), g*sign(sin(p))*sqrt(abs(fy(c+hw*cos(p)))), p, 0, 2*pi)
um = g*k*(xm-c)
lSeg = Segment((-g*L/2, 0), (g*L/2, 0))
wSeg = Segment((um, -g*W/2), (um, g*W/2))
auxSeg = Segment((um+0.35, -g*auxLen/2), (um+0.35, g*auxLen/2))
LWmeasured = Length(lSeg)/Length(wSeg)
auxMeasured = Length(lSeg)/Length(auxSeg)
rPi = (Length(b1)+Length(b2))/Length(hSeg)
rPhi = Length(sSeg)/Length(aSeg)
errPi = abs(rPi-pi)/pi
errPhi = abs(rPhi-phi)/phi
ratP = {3, 8, 11, 14, 19, 25}
ratQ = {2, 5, 7, 9, 12, 16}
errPiList = Sequence(abs(2*Element(ratP, i)/Element(ratQ, i) - pi)/pi, i, 1, 6)
errPhiList = Sequence(abs(sqrt(1+4/(Element(ratP, i)/Element(ratQ, i))^2) - phi)/phi, i, 1, 6)
pickP = Element(ratP, pick)
pickQ = Element(ratQ, pick)
pickPi = 2*pickP/pickQ
pickPhi = sqrt(1+4/(pickP/pickQ)^2)
barsPi = Sequence(Segment((3*i-0.5, -6), (3*i-0.5, -6+Min(5, 100*Element(errPiList, i)))), i, 1, 6)
barsPhi = Sequence(Segment((3*i+0.5, -6), (3*i+0.5, -6+Min(5, 100*Element(errPhiList, i)))), i, 1, 6)
barLabels = Sequence(Text(Element(ratP, i) + ":" + Element(ratQ, i), (3*i-0.9, -7.2)), i, 1, 6)
pickMark = Segment((3*pick-1.2, -6.4), (3*pick+1.2, -6.4))
titleText = Text(If(langEN, "Pyramid 11:7 | G6 · geometric extensions", "Piramida 11:7 | G6 · rozszerzenia geometryczne"), (0.7, 28.4))
lessonText = Text(If(lesson == 1, If(langEN, "1. The golden egg ratio limit", "1. Granica proporcji złotego jaja"), If(lesson == 2, If(langEN, "2. Inscribed sphere and the golden section", "2. Kula wpisana i złoty podział"), If(langEN, "3. Why the 11:7 ratio?", "3. Dlaczego proporcja 11:7?"))), (0.7, 11.7))
plainText = Text(If(lesson == 1, If(limitSA > phi, If(langEN, "Moving the section brings us closer to a limit. The golden ratio lies slightly below it.", "Przesuwając przekrój, zbliżamy się do granicy. Złota proporcja leży odrobinę poniżej niej."), If(langEN, "At this slope the limit S/A is not above φ.", "Przy tym nachyleniu granica S/A nie leży powyżej φ.")), If(lesson == 2, If(langEN, "The same φ family as S/A, not a new confirmation.", "Ta sama rodzina φ co S/A, nie nowe potwierdzenie."), If(langEN, "One small fraction near both π and φ; one slope.", "Mały ułamek blisko π i φ naraz; jedno nachylenie."))), (0.7, 10.5))
v1Text = Text(If(lesson == 1, If(eggOK, "L/W = " + round(LW, 12), If(langEN, "No oval: z0² > 4t is required", "Brak owalu: potrzeba z0² > 4t") + " (z0 > " + round(2*sqrt(t), 6) + ")"), If(lesson == 2, "r = " + round(r, 12) + " ; VI/OI = " + round(VIoverOI, 12), "2B/h = " + round(rPi, 12) + " ; " + round(100*errPi, 9) + " %")), (0.7, 9.3))
v2Text = Text(If(lesson == 1, "S/A = √(1+t²) = " + round(limitSA, 12), If(lesson == 2, "MT = A = " + round(MT, 9) + " ; VT = S − A = " + round(VT, 9), "S/A = " + round(rPhi, 12) + " ; " + round(100*errPhi, 9) + " %")), (0.7, 8.1))
v3Text = Text(If(lesson == 1, "φ = " + round(phi, 12) + If(eggOK, " ; " + round(100*errLW, 9) + " %", ""), If(lesson == 2, "MT/VT = " + round(MToverVT, 12) + " ; " + round(100*errMTVT, 9) + " %", pickP + ":" + pickQ + " → 2B/h = " + round(pickPi, 6) + " ; S/A = " + round(pickPhi, 6))), (0.7, 6.9))
v4Text = Text(If(lesson == 1, "z0 = " + round(z0, 2) + " ; L = " + If(eggOK, round(L, 9), "—") + " ; W = " + If(eggOK, round(W, 9), "—"), If(lesson == 2, If(langEN, "Distance from I to base and 4 faces = r", "Odległość I od podstawy i 4 ścian = r"), If(langEN, "Ordered by denominator; not a ranking.", "Kolejność wg mianownika; to nie ranking."))), (0.7, 5.7))
stepText = Text(If(lesson == 2, If(sphereStep == 1, If(langEN, "Step 1: sphere and axial section", "Krok 1: kula i przekrój osiowy"), If(sphereStep == 2, If(langEN, "Step 2: OI = r, IV = h − r", "Krok 2: OI = r, IV = h − r"), If(sphereStep == 3, If(langEN, "Step 3: T, MT = MO = A, VT = S − A", "Krok 3: T, MT = MO = A, VT = S − A"), If(langEN, "Step 4: VI/OI = S/A ≈ φ, MT/VT ≈ φ", "Krok 4: VI/OI = S/A ≈ φ, MT/VT ≈ φ")))), If(lesson == 1, If(langEN, "Dashed: width L/(S/A); W is always smaller.", "Linia przerywana: szerokość L/(S/A); W jest zawsze mniejsze."), If(langEN, "Bars: error vs π (left) and φ (right), in %, capped at 5%.", "Słupki: błąd względem π (lewy) i φ (prawy), w %, obcięte na 5%."))), (0.7, 4.5))
